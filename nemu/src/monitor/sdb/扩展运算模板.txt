static struct rule {
  const char *regex;
  int token_type;
} rules[] = {
  // 多字符运算符优先（必须放在单字符运算符之前，避免误匹配）
  {"0[xX][0-9a-fA-F]+u", TK_HEX},      // 十六进制数 + 'u' 后缀，如 0x1234u
  {"0[xX][0-9a-fA-F]+",  TK_HEX},      // 普通十六进制数，如 0x8000, 0Xabcd
  {"[0-9]+u",            TK_DEC},      // 十进制数 + 'u' 后缀，如 42u
  {"[0-9]+",             TK_DEC},      // 普通十进制数，如 123
  {"\\$[a-zA-Z0-9_]+",   TK_REG},      // 寄存器，以 $ 开头，如 $eax, $pc

  // 多字符运算符
  {"==",                 TK_EQ},       // 等于运算符 ==
  {"!=",                 TK_NEQ},      // 不等于运算符 !=
  {"&&",                 TK_AND},      // 逻辑与 &&
  {"<=",                 TK_LE},       // 小于等于 <=
  {">=",                 TK_GE},       // 大于等于 >=
  {"<<",                 TK_SHL},      // 左移 <<
  {">>",                 TK_SHR},      // 右移 >>

  // 单字符运算符（部分字符在正则中有特殊含义，需转义）
  {"&",                  TK_BIT_AND},  // 按位与 &
  {"\\|",                TK_BIT_OR},   // 按位或 |
  {"\\^",                TK_BIT_XOR},  // 按位异或 ^
  {"<",                  TK_LT},       // 小于 <
  {">",                  TK_GT},       // 大于 >
  {"\\+",                TK_ADD},      // 加法 +
  {"\\-",                TK_SUB},      // 减法 -
  {"\\*",                TK_MUL},      // 乘法 *
  {"\\/",                TK_DIV},      // 除法 /
  {"\\(",                TK_LPAREN},   // 左括号 (
  {"\\)",                TK_RPAREN},   // 右括号 )
  {" +",                 TK_NOTYPE},   // 空格
};
